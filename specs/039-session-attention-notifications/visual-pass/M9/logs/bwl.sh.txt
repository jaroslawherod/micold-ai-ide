#!/bin/bash
# bwl.sh <tag> <pre> <minx,miny>: Wayland client; pre = none | min
. /tmp/claude-1000/m9/xe.sh
C=a0082f83-376a-4ff8-9976-727f40a801d0; t=$1; pre=$2; mb=$3
D() { "${PENV[@]}" python3 $P/drive.py "$@"; }
[ "$pre" = min ] && D click:$mb sleep:1.5 > $P/logs/$t-prep.log 2>&1
"${PENV[@]}" python3 $P/pollw.py $P/logs/$t-focus.txt 22 &
PP=$!
sleep 0.7
D shot:$P/shots/$t-1-before.png "run:$P/hook.sh $C UserPromptSubmit" sleep:1.5 "run:$P/hook.sh $C Stop" sleep:2.5 shot:$P/shots/$t-2-banner.png move:640,90 sleep:0.4 press sleep:0.6 shot:$P/shots/$t-3-after0.6.png sleep:1.4 shot:$P/shots/$t-4-after2.png sleep:3.5 shot:$P/shots/$t-5-after6.png > $P/logs/$t.log 2>&1
wait $PP
grep "press" $P/logs/$t.log | cut -c1-80
cat $P/logs/$t-focus.txt | cut -c1-250
